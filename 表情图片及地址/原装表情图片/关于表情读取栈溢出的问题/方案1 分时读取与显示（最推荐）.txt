#include "stm32f10x.h"
#include "Delay.h"
#include "OLED.h"
#include "W25Q64.h"

// 单缓冲区（1024字节）
__attribute__((section(".ccmram"))) uint8_t g_imageBuffer[1024];

// 图片地址数组
const uint32_t IMAGE_ADDRESSES[4] = {
    0x000000,  // 图片1地址
    0x001000,  // 图片2地址
    0x002000,  // 图片3地址
    0x003000   // 图片4地址
};

int main(void)
{
    // 初始化
    SystemInit();
    Delay_Init();
    OLED_Init();
    OLED_Clear();
    W25Q64_Init();
    
    // 显示启动信息
    OLED_ShowString(0, 0, "Image Display", 16, 1);
    OLED_Refresh();
    
    // 主循环
    while (1)
    {
        // 循环显示四张图片
        for (int i = 0; i < 4; i++)
        {
            // 从Flash读取当前图片
            W25Q64_ReadData(IMAGE_ADDRESSES[i], g_imageBuffer, sizeof(g_imageBuffer));
            
            // 显示图片
            OLED_Clear();
            OLED_ShowPicture(0, 0, 128, 64, g_imageBuffer, 1);
            OLED_Refresh();
            
            // 延时
            Delay_ms(1000);
        }
    }
}
